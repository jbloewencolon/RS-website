# Relational Sovereignty — crawler policy
# Opacity is owed to the crawler and the market, never to the disabled reader.
# Everything disallowed here remains fully available to assistive technology,
# to any person who asks, and to anyone who forks the repository.

# The reading material is public. Find it, index it, pass it on.
# BUG-03 (2026-08-08): every path below moved from a flat *.dc.html
# filename to a clean directory URL. The old paths still resolve — each
# is now a thin, noindex'd redirect page pointing here — so an already
# indexed or bookmarked *.dc.html link still lands on the right content.
User-agent: *

Allow: /archive/
Allow: /behind-the-scenes/

# Resources is the one page on this site that should be indexed by
# everything, no exceptions, including the training crawlers refused
# below. A person who needs this page matters more than the extraction
# refusal.
Allow: /resources/

# The workshop is not for the index. What a person marks about their own
# relationships is not material for a register, a snapshot, or a training set.
Disallow: /practise/

# Contributor material is governed by the terms each contributor chose,
# not by discovery. Ask, or read Behind the Scenes.
Disallow: /contribute/

Sitemap: https://relationalsovereignty.com/sitemap.xml
